[CmdletBinding()]
param(
    [string]$PackageRoot = $PSScriptRoot,
    [switch]$ValidationOnly,
    [string]$TestHarnessConfig = ''
)

$ErrorActionPreference = 'Stop'
$PackageVersion = '1.4.4'

function Get-NormalizedPackageRoot {
    param([AllowNull()][string]$Path)

    if ([string]::IsNullOrWhiteSpace($Path)) {
        $Path = $PSScriptRoot
    }

    $Path = $Path.Trim().Trim('"')
    if ([string]::IsNullOrWhiteSpace($Path)) {
        throw 'Der Paketpfad ist leer.'
    }

    try {
        $full = [System.IO.Path]::GetFullPath($Path)
    } catch {
        throw ("Der Paketpfad ist ungueltig: " + $_.Exception.Message)
    }

    if (-not (Test-Path -LiteralPath $full -PathType Container)) {
        throw "Der Paketpfad wurde nicht gefunden: $full"
    }

    return (Get-Item -LiteralPath $full -ErrorAction Stop).FullName
}

$PackageRoot = Get-NormalizedPackageRoot -Path $PackageRoot

$TestHarness = $null
if (-not [string]::IsNullOrWhiteSpace($TestHarnessConfig)) {
    $TestHarnessConfig = [System.IO.Path]::GetFullPath($TestHarnessConfig.Trim().Trim('"'))
    if (-not (Test-Path -LiteralPath $TestHarnessConfig -PathType Leaf)) {
        throw "Test-Harness-Konfiguration wurde nicht gefunden: $TestHarnessConfig"
    }
    $TestHarness = Get-Content -LiteralPath $TestHarnessConfig -Raw -ErrorAction Stop | ConvertFrom-Json
}

function Get-TestHarnessValue {
    param([string]$Name)
    if ($null -eq $TestHarness) { return $null }
    $property = $TestHarness.PSObject.Properties[$Name]
    if ($null -eq $property) { return $null }
    return $property.Value
}
$harnessBase = Get-TestHarnessValue -Name 'base'
$Base = if (-not [string]::IsNullOrWhiteSpace([string]$harnessBase)) {
    [System.IO.Path]::GetFullPath([string]$harnessBase)
} elseif (-not [string]::IsNullOrWhiteSpace($env:VEYON_UPDATER_BASE)) {
    [System.IO.Path]::GetFullPath(([string]$env:VEYON_UPDATER_BASE).Trim().Trim('"'))
} else {
    'C:\ProgramData\Veyon\Update'
}
$MainScriptName = 'Veyon-AutoUpdate.ps1'
$MainScriptSource = Join-Path $PackageRoot $MainScriptName
$MainScriptPath = Join-Path $Base $MainScriptName

$harnessTaskName = Get-TestHarnessValue -Name 'taskName'
$harnessHelperTaskName = Get-TestHarnessValue -Name 'helperTaskName'
$TaskName = if (-not [string]::IsNullOrWhiteSpace([string]$harnessTaskName)) {
    [string]$harnessTaskName
} elseif ($env:VEYON_UPDATER_TASK_NAME) {
    [string]$env:VEYON_UPDATER_TASK_NAME
} else {
    'Veyon AutoUpdate (winget)'
}
$HelperTaskName = if (-not [string]::IsNullOrWhiteSpace([string]$harnessHelperTaskName)) {
    [string]$harnessHelperTaskName
} elseif ($env:VEYON_UPDATER_HELPER_TASK_NAME) {
    [string]$env:VEYON_UPDATER_HELPER_TASK_NAME
} else {
    'Veyon AutoUpdate Setup Helper'
}

function Add-TestHarnessArgument {
    param([string]$Arguments)
    if (-not [string]::IsNullOrWhiteSpace($TestHarnessConfig)) {
        return ($Arguments + (' -TestHarnessConfig "{0}"' -f $TestHarnessConfig))
    }
    return $Arguments
}

function Test-IsAdministrator {
    try {
        $identity = [Security.Principal.WindowsIdentity]::GetCurrent()
        $principal = New-Object Security.Principal.WindowsPrincipal($identity)
        return $principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
    } catch {
        return $false
    }
}

function Get-NativeWindowsPowerShell {
    $system32 = Join-Path $env:SystemRoot 'System32\WindowsPowerShell\v1.0\powershell.exe'
    if ([Environment]::Is64BitOperatingSystem -and -not [Environment]::Is64BitProcess) {
        $sysnative = Join-Path $env:SystemRoot 'Sysnative\WindowsPowerShell\v1.0\powershell.exe'
        if (Test-Path -LiteralPath $sysnative -PathType Leaf) { return $sysnative }
    }
    return $system32
}
