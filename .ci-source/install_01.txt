    $arguments = '-NoProfile -ExecutionPolicy Bypass -File "{0}" {1}' -f $MainScriptPath,$ModeArguments
    $arguments = Add-TestHarnessArgument -Arguments $arguments
    $task = New-SystemTaskDefinition -Arguments $arguments -TriggerType Startup

    try {
        Register-ScheduledTask -TaskName $HelperTaskName -InputObject $task -Force | Out-Null
        $startedAt = Get-Date
        Start-ScheduledTask -TaskName $HelperTaskName

        $deadline = (Get-Date).AddSeconds($TimeoutSeconds)
        $hasRun = $false
        $current = $null
        $info = $null
        do {
            Start-Sleep -Milliseconds 500
            $current = Get-ScheduledTask -TaskName $HelperTaskName -ErrorAction Stop
            $info = Get-ScheduledTaskInfo -TaskName $HelperTaskName -ErrorAction Stop
            $hasRun = ($info.LastRunTime -ge $startedAt.AddSeconds(-2))
        } while ((-not $hasRun -or $current.State -eq 'Running') -and (Get-Date) -lt $deadline)

        if (-not $hasRun) {
            throw 'SYSTEM-Hilfstask wurde nicht gestartet.'
        }
        if ($current.State -eq 'Running') {
            try { Stop-ScheduledTask -TaskName $HelperTaskName -ErrorAction SilentlyContinue } catch {}
            throw "SYSTEM-Hilfstask lief laenger als $TimeoutSeconds Sekunden."
        }

        return [uint32]$info.LastTaskResult
    }
    finally {
        Remove-TaskQuietly -Name $HelperTaskName
    }
}
