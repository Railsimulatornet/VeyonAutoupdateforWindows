        [ref]$tokens,
        [ref]$parseErrors
    ) | Out-Null

    if ($parseErrors -and $parseErrors.Count -gt 0) {
        $first = $parseErrors | Select-Object -First 1
        throw ("PowerShell-Syntaxfehler in " + $Path + ': ' + $first.Message)
    }
}

function Remove-TaskQuietly {
    param([string]$Name)
    try {
        Unregister-ScheduledTask -TaskName $Name -Confirm:$false -ErrorAction SilentlyContinue
    } catch {}
}

function New-SystemTaskDefinition {
    param(
        [Parameter(Mandatory=$true)][string]$Arguments,
        [ValidateSet('Startup','Logon')][string]$TriggerType = 'Startup',
        [switch]$WithDelay,
        [string]$Description = ''
    )

    $powerShellExe = Get-NativeWindowsPowerShell
    if (-not (Test-Path -LiteralPath $powerShellExe -PathType Leaf)) {
        throw "Windows PowerShell wurde nicht gefunden: $powerShellExe"
    }

    $action = New-ScheduledTaskAction -Execute $powerShellExe -Argument $Arguments
    if ($TriggerType -eq 'Logon') {
        $trigger = New-ScheduledTaskTrigger -AtLogOn
    } else {
        $trigger = New-ScheduledTaskTrigger -AtStartup
    }
    if ($WithDelay) {
        $trigger.Delay = 'PT2M'
    }

    $principal = New-ScheduledTaskPrincipal -UserId 'SYSTEM' -LogonType ServiceAccount -RunLevel Highest
    $settings = New-ScheduledTaskSettingsSet `
        -AllowStartIfOnBatteries `
        -DontStopIfGoingOnBatteries `
        -StartWhenAvailable `
        -MultipleInstances IgnoreNew `
        -ExecutionTimeLimit (New-TimeSpan -Hours 2)

    return New-ScheduledTask -Action $action -Trigger $trigger -Principal $principal -Settings $settings -Description $Description
}

function Invoke-SystemUpdaterMode {
    param(
        [Parameter(Mandatory=$true)][string]$ModeArguments,
        [int]$TimeoutSeconds = 300
    )

    Remove-TaskQuietly -Name $HelperTaskName
    $arguments = '-NoProfile -ExecutionPolicy Bypass -File "{0}" {1}' -f $MainScriptPath,$ModeArguments
    $arguments = Add-TestHarnessArgument -Arguments $arguments
    $task = New-SystemTaskDefinition -Arguments $arguments -TriggerType Startup -Description 'Veyon AutoUpdate SYSTEM helper'

    try {
        Register-ScheduledTask -TaskName $HelperTaskName -InputObject $task -Force | Out-Null
        $startedAt = Get-Date
        Start-ScheduledTask -TaskName $HelperTaskName

        $deadline = (Get-Date).AddSeconds($TimeoutSeconds)
        $hasRun = $false
        $current = $null
        $info = $null
        do {
            Start-Sleep -Milliseconds 500
            $current = Get-ScheduledTask -TaskName $HelperTaskName -ErrorAction Stop
            $info = Get-ScheduledTaskInfo -TaskName $HelperTaskName -ErrorAction Stop
            $hasRun = ($info.LastRunTime -ge $startedAt.AddSeconds(-2))
        } while ((-not $hasRun -or $current.State -eq 'Running') -and (Get-Date) -lt $deadline)

        if (-not $hasRun) {
            throw 'SYSTEM-Hilfstask wurde nicht gestartet.'
        }
        if ($current.State -eq 'Running') {
            try { Stop-ScheduledTask -TaskName $HelperTaskName -ErrorAction SilentlyContinue } catch {}
            throw "SYSTEM-Hilfstask lief laenger als $TimeoutSeconds Sekunden."
        }

        return [uint32]$info.LastTaskResult
    }
    finally {
        Remove-TaskQuietly -Name $HelperTaskName
    }
}
