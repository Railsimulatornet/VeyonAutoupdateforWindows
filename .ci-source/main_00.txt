[CmdletBinding()]
param(
    [switch]$Testlauf,
    [switch]$SkipSelfUpdate,
    [switch]$ConfigurePinOnly,
    [switch]$RemovePinOnly,
    [string]$TestHarnessConfig = ''
)

$ErrorActionPreference = 'Stop'

$TestHarness = $null
if (-not [string]::IsNullOrWhiteSpace($TestHarnessConfig)) {
    try {
        $TestHarnessConfig = [System.IO.Path]::GetFullPath($TestHarnessConfig.Trim().Trim('"'))
        if (-not (Test-Path -LiteralPath $TestHarnessConfig -PathType Leaf)) {
            throw "Test-Harness-Konfiguration wurde nicht gefunden: $TestHarnessConfig"
        }
        $TestHarness = Get-Content -LiteralPath $TestHarnessConfig -Raw -ErrorAction Stop | ConvertFrom-Json
    } catch {
        throw ("Test-Harness-Konfiguration ist ungueltig: " + $_.Exception.Message)
    }
}

function Get-TestHarnessValue {
    param([string]$Name)
    if ($null -eq $TestHarness) { return $null }
    $property = $TestHarness.PSObject.Properties[$Name]
    if ($null -eq $property) { return $null }
    return $property.Value
}

# TLS 1.2 fuer Windows PowerShell 5.1 sicherstellen.
try {
    [Net.ServicePointManager]::SecurityProtocol =
        [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
} catch {}

$ScriptVersion = [version]'1.4.4.0'
$DefaultSelfUpdateManifestUrl = 'https://raw.githubusercontent.com/Railsimulatornet/VeyonAutoupdateforWindows/main/update-manifest.json'
$harnessManifestUrl = Get-TestHarnessValue -Name 'selfUpdateManifestUrl'
$SelfUpdateManifestUrl = if (-not [string]::IsNullOrWhiteSpace([string]$harnessManifestUrl)) {
    [string]$harnessManifestUrl
} elseif (-not [string]::IsNullOrWhiteSpace($env:VEYON_UPDATER_SELFUPDATE_MANIFEST_URL)) {
    [string]$env:VEYON_UPDATER_SELFUPDATE_MANIFEST_URL
} else {
    $DefaultSelfUpdateManifestUrl
}
$TestMode = ($null -ne $TestHarness) -or ([string]$env:VEYON_UPDATER_TEST_MODE -eq '1')

$PkgId   = 'VeyonSolutions.Veyon'
$harnessBase = Get-TestHarnessValue -Name 'base'
$Base = if (-not [string]::IsNullOrWhiteSpace([string]$harnessBase)) {
    [System.IO.Path]::GetFullPath([string]$harnessBase)

$de = [System.Globalization.CultureInfo]::GetCultureInfo('de-DE')

function Write-Log {
    param([string]$Text)
    $stamp = (Get-Date).ToString('dd.MM.yyyy HH:mm:ss', $de)
    Add-Content -LiteralPath $Log -Value "[$stamp] $Text" -Encoding UTF8
}

function Normalize-Version {
    param([AllowNull()][version]$Version)

    if ($null -eq $Version) { return $null }

    $build = $Version.Build
    if ($build -lt 0) { $build = 0 }

    $revision = $Version.Revision
    if ($revision -lt 0) { $revision = 0 }

    try {
        return [version]("{0}.{1}.{2}.{3}" -f
            $Version.Major, $Version.Minor, $build, $revision)
    } catch {
        return $null
    }
}

function Parse-VersionText {
    param([AllowNull()][string]$Text)

    if ([string]::IsNullOrWhiteSpace($Text)) { return $null }
    if ($Text -match '(\d+(?:\.\d+){1,3})') {
        try {
            return Normalize-Version -Version ([version]$matches[1])
        } catch {
            return $null
        }
    }
    return $null
}

function Test-CanonicalScriptPath {
    if ([string]::IsNullOrWhiteSpace($PSCommandPath)) { return $false }
    try {
        $current = [System.IO.Path]::GetFullPath($PSCommandPath)
        $target  = [System.IO.Path]::GetFullPath($MainScriptPath)
        return [string]::Equals($current, $target, [System.StringComparison]::OrdinalIgnoreCase)
    } catch {
        return $false
    }
}

function Invoke-SelfUpdate {
    if ($SkipSelfUpdate) {
        Write-Log 'Skript-Self-Update uebersprungen (-SkipSelfUpdate).'
        return
    }

    if ($ConfigurePinOnly -or $RemovePinOnly) {
        return
    }

    if (-not (Test-CanonicalScriptPath)) {
        Write-Log 'Skript-Self-Update uebersprungen: Skript laeuft nicht unter dem produktiven Zielpfad.'
        return
    }

    $tmp = Join-Path $Base 'Veyon-AutoUpdate.selfupdate.new.ps1'
    $bak = Join-Path $Base 'Veyon-AutoUpdate.selfupdate.bak.ps1'
    $headers = @{
        'User-Agent'    = 'Veyon-AutoUpdate'
        'Cache-Control' = 'no-cache'
        'Pragma'        = 'no-cache'
    }

    try {
        $manifestAllowed = ($SelfUpdateManifestUrl -match '^https://raw\.githubusercontent\.com/Railsimulatornet/VeyonAutoupdateforWindows/') -or
            ($TestMode -and $SelfUpdateManifestUrl -match '^https?://(?:127\.0\.0\.1|localhost)(?::\d+)?/')
        if (-not $manifestAllowed) {
            throw 'Self-Update-Manifest-URL ist nicht erlaubt.'
        }

        $nonce = [DateTimeOffset]::UtcNow.ToUnixTimeSeconds()
        $separator = if ($SelfUpdateManifestUrl.Contains('?')) { '&' } else { '?' }
        $manifestUri = $SelfUpdateManifestUrl + $separator + 't=' + $nonce
        $manifest = Invoke-RestMethod -Uri $manifestUri -Headers $headers -UseBasicParsing -TimeoutSec 30

        if ($null -eq $manifest -or [int]$manifest.schemaVersion -ne 1) {
            throw 'Self-Update-Manifest fehlt oder hat eine unbekannte Schema-Version.'
        }

        $remoteVersion = Parse-VersionText ([string]$manifest.version)
        $localVersion  = Normalize-Version -Version $ScriptVersion
        if ($null -eq $remoteVersion) {
            throw 'Remote-Skriptversion im Self-Update-Manifest ist ungueltig.'
        }

        if ($remoteVersion -le $localVersion) {
            Write-Log ("Skript-Self-Update: bereits aktuell (" + $localVersion + ").")
            return
        }
